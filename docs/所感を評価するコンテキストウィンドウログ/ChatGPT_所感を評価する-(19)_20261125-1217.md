def extract_video_id(url: str) -> str | None:
    """
    対応するYouTubeのURLから動画IDを抽出する。

    対応するURL形式:
    - 標準URL: https://www.youtube.com/watch?v=xxxxxxxxxxx

    if url.startswith("//"):

        if parsed_url.scheme.lower() not in {"http", "https"}:

            return None
